<script setup>
import { computed } from 'vue'

definePageMeta({
  layout: 'default'
})

const route = useRoute()
const router = useRouter()
const { $api } = useNuxtApp()

const userAnswerId = route.params.userAnswerId

const { data: resultResponse, pending: loadingResult } = await useAsyncData(
  `result-summary-${userAnswerId}`,
  () => $api(`/api/v1/user-answers/explain/${userAnswerId}`, { method: 'GET' }),
  { server: false }
)

const explanations = computed(() => resultResponse.value?.data || [])

const totalQuestions = computed(() => explanations.value.length)
const correctCount   = computed(() => explanations.value.filter(item => item?.status === 1).length)
const wrongCount     = computed(() => explanations.value.filter(item => item?.userAnswerTitle && item?.status === 0).length)
const skippedCount   = computed(() => explanations.value.filter(item => !item?.userAnswerTitle).length)
const totalPoint     = computed(() => explanations.value.reduce((sum, item) => sum + (item?.point || 0), 0))
const accuracy       = computed(() => {
  if (totalQuestions.value === 0) return 0
  return Math.round((correctCount.value / totalQuestions.value) * 100)
})

const questionBankId = computed(() => explanations.value[0]?.questionBankId || null)

const twkScore = computed(() => explanations.value.filter(q => q.mainTopic?.toUpperCase().includes('TWK')).reduce((sum, q) => sum + (q.point || 0), 0))
const tiuScore = computed(() => explanations.value.filter(q => q.mainTopic?.toUpperCase().includes('TIU')).reduce((sum, q) => sum + (q.point || 0), 0))
const tkpScore = computed(() => explanations.value.filter(q => q.mainTopic?.toUpperCase().includes('TKP')).reduce((sum, q) => sum + (q.point || 0), 0))

const { data: bankResponse } = await useAsyncData(
  `bank-detail-${userAnswerId}`,
  () => {
    if (!questionBankId.value) return Promise.resolve(null)
    return $api(`/api/v1/question-banks/${questionBankId.value}`, { method: 'GET' })
  },
  { server: false, lazy: true, watch: [questionBankId] }
)

const bankData    = computed(() => bankResponse.value?.data || null)
const targetTwk   = computed(() => bankData.value?.passingGradeTwk   || 0)
const targetTiu   = computed(() => bankData.value?.passingGradeTiu   || 0)
const targetTkp   = computed(() => bankData.value?.passingGradeTkp   || 0)
const targetTotal = computed(() => bankData.value?.passingGradeTotal || 0)

const isSkd = computed(() => targetTwk.value > 0 || targetTiu.value > 0 || targetTkp.value > 0 || targetTotal.value > 0)

const isPassed = computed(() => {
  if (!isSkd.value) return null
  const passTwk   = targetTwk.value === 0   || twkScore.value >= targetTwk.value
  const passTiu   = targetTiu.value === 0   || tiuScore.value >= targetTiu.value
  const passTkp   = targetTkp.value === 0   || tkpScore.value >= targetTkp.value
  const passTotal = targetTotal.value === 0 || totalPoint.value >= targetTotal.value
  return passTwk && passTiu && passTkp && passTotal
})

const getProgress = (score, target) => {
  if (!target || target === 0) return 100
  return Math.min(100, Math.max(0, (score / target) * 100))
}

const { data: leaderboardResponse, pending: loadingLeaderboard } = await useAsyncData(
  `leaderboard-${userAnswerId}`,
  () => $api(`/api/v1/user-answers/${userAnswerId}/leaderboard`, { method: 'GET' }),
  { server: false }
)

const leaderboard      = computed(() => leaderboardResponse.value?.data || {})
const rankings         = computed(() => leaderboard.value?.top100 || [])
const userRank         = computed(() => leaderboard.value?.userRank || null)
const totalParticipants = computed(() => leaderboard.value?.totalParticipants || 0)

const isUserInTop100 = computed(() => rankings.value.some(r => r.isCurrentUser))
const hasLeaderboardData = computed(() => rankings.value.length > 0 || userRank.value !== null)

// Rank color — warna medali, tidak ikut theme
const getRankColor = (index) => {
  if (index === 0) return 'bg-gradient-to-br from-yellow-400 to-yellow-600 text-white'
  if (index === 1) return 'bg-gradient-to-br from-gray-300 to-gray-500 text-white'
  if (index === 2) return 'bg-gradient-to-br from-orange-400 to-orange-600 text-white'
  return 'bg-tryout-hover text-tryout-primary border border-tryout-border'
}

const getRankBadge = (index) => {
  if (index === 0) return '🥇'
  if (index === 1) return '🥈'
  if (index === 2) return '🥉'
  return ''
}

const goToPembahasanDetail = () => router.push(`/pembahasan/${userAnswerId}/detail`)
const goBack = () => router.back()
</script>

<template>
  <!-- bg-tryout-bg → #f0f4ff (light) / #0f172a (dark) otomatis dari main.css -->
  <main class="min-h-screen py-8 bg-tryout-bg">

    <!-- Loading -->
    <div v-if="loadingResult" class="flex items-center justify-center min-h-screen">
      <div class="text-center">
        <div class="animate-spin rounded-full h-16 w-16 border-4 mx-auto
                    border-tryout-border border-t-tryout-primary"></div>
        <p class="mt-6 font-bold text-tryout-muted">Menghitung hasil ujian...</p>
      </div>
    </div>

    <div v-else class="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">

      <!-- Page header -->
      <div class="mb-8">
        <NuxtLink
          :to="`/tryout/${questionBankId}/info`"
          class="mb-6 flex items-center gap-2 text-sm font-bold transition-colors group
                 text-tryout-muted hover:text-tryout-text"
        >
          <span class="p-1.5 rounded-lg transition-all
                       bg-tryout-surface border border-tryout-border
                       group-hover:bg-tryout-hover">
            <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M15 19l-7-7 7-7" />
            </svg>
          </span>
          Kembali ke tryout
        </NuxtLink>

        <!-- Judul — gradient tetap pakai hardcode karena dekoratif, bukan theme -->
        <h1 class="text-4xl font-extrabold bg-gradient-to-r from-blue-600 to-purple-600 bg-clip-text text-transparent mb-2">
          🎉 Tryout Selesai!
        </h1>
        <p class="font-medium text-tryout-muted">Lihat hasil dan ranking Anda di bawah ini</p>
      </div>

      <div class="grid grid-cols-1 lg:grid-cols-3 gap-8">

        <!-- ── Kolom Kiri ── -->
        <div class="lg:col-span-2 space-y-6">

          <!-- Card Ringkasan Hasil -->
          <div class="rounded-3xl shadow-xl p-8 bg-tryout-surface border border-tryout-border">
            <div class="flex flex-col sm:flex-row sm:items-center justify-between gap-4 mb-6">
              <h2 class="text-2xl font-bold text-tryout-text">📊 Ringkasan Hasil</h2>
              <div class="flex items-center gap-2 px-5 py-2.5 rounded-full border shadow-sm
                          bg-tryout-primary/10 border-tryout-primary/20">
                <svg class="w-5 h-5 text-tryout-primary" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M5 3v4M3 5h4M6 17v4m-2-2h4m5-16l2.286 6.857L21 12l-5.714 2.143L13 21l-2.286-6.857L5 12l5.714-2.143L13 3z" />
                </svg>
                <span class="font-extrabold text-tryout-primary">{{ totalPoint }} Poin</span>
              </div>
            </div>

            <!-- SKD: banner lulus/gagal + progress bar TWK/TIU/TKP -->
            <template v-if="isSkd">
              <div
                class="mb-6 p-4 rounded-2xl flex items-center gap-4"
                :class="isPassed
                  ? 'bg-emerald-500/10 border border-emerald-500/20'
                  : 'bg-red-500/10 border border-red-500/20'"
              >
                <div class="w-12 h-12 rounded-full flex items-center justify-center text-white text-xl shadow-sm shrink-0"
                     :class="isPassed ? 'bg-emerald-500' : 'bg-red-500'">
                  {{ isPassed ? '🎉' : '❌' }}
                </div>
                <div>
                  <h3 class="font-extrabold text-lg"
                      :class="isPassed ? 'text-emerald-500' : 'text-red-500'">
                    {{ isPassed ? 'LULUS PASSING GRADE' : 'BELUM LULUS' }}
                  </h3>
                  <p class="text-sm font-medium"
                     :class="isPassed ? 'text-emerald-500' : 'text-red-500'">
                    {{ isPassed ? 'Selamat! Nilai Anda memenuhi ambang batas minimal.' : 'Nilai Anda belum memenuhi ambang batas (Passing Grade).' }}
                  </p>
                </div>
              </div>

              <!-- Grid TWK / TIU / TKP -->
              <div class="grid grid-cols-1 md:grid-cols-3 gap-4 mb-8">
                <div
                  v-for="({ label, score, target }) in [
                    { label: 'TWK', score: twkScore, target: targetTwk },
                    { label: 'TIU', score: tiuScore, target: targetTiu },
                    { label: 'TKP', score: tkpScore, target: targetTkp },
                  ]"
                  :key="label"
                  class="relative overflow-hidden p-4 rounded-2xl bg-tryout-hover border border-tryout-border"
                >
                  <div class="absolute top-0 left-0 right-0 h-1"
                       :class="score >= target ? 'bg-emerald-500' : 'bg-red-500'"></div>
                  <div class="flex justify-between items-end mb-2">
                    <span class="font-black text-sm text-tryout-text">{{ label }}</span>
                    <div class="flex items-baseline gap-1">
                      <span class="text-xl font-black"
                            :class="score >= target ? 'text-emerald-500' : 'text-red-500'">
                        {{ score }}
                      </span>
                      <span class="text-xs font-bold text-tryout-subtle">/ {{ target }}</span>
                    </div>
                  </div>
                  <div class="w-full h-1.5 rounded-full overflow-hidden bg-tryout-skeleton">
                    <div
                      class="h-full rounded-full transition-all duration-1000 ease-out"
                      :class="score >= target ? 'bg-emerald-500' : 'bg-red-500'"
                      :style="{ width: getProgress(score, target) + '%' }"
                    ></div>
                  </div>
                </div>
              </div>
            </template>

            <!-- Stat cards: Total / Benar / Salah / Akurasi -->
            <!-- Warna kartu stat tetap pakai warna kategori (biru/hijau/merah/oranye) bukan theme -->
            <div class="grid grid-cols-2 md:grid-cols-4 gap-4 mb-8">
              <div class="bg-gradient-to-br from-blue-50 to-blue-100 dark:from-blue-900/20 dark:to-blue-900/30 rounded-2xl p-5 text-center border border-blue-200 dark:border-blue-700/40 shadow-sm">
                <div class="text-3xl font-extrabold text-blue-600 dark:text-blue-400">{{ totalQuestions }}</div>
                <div class="text-xs mt-1 font-bold uppercase tracking-wider text-blue-800 dark:text-blue-300">Total Soal</div>
              </div>
              <div class="bg-gradient-to-br from-emerald-50 to-emerald-100 dark:from-emerald-900/20 dark:to-emerald-900/30 rounded-2xl p-5 text-center border border-emerald-200 dark:border-emerald-700/40 shadow-sm">
                <div class="text-3xl font-extrabold text-emerald-600 dark:text-emerald-400">{{ correctCount }}</div>
                <div class="text-xs mt-1 font-bold uppercase tracking-wider text-emerald-800 dark:text-emerald-300">Benar</div>
              </div>
              <div class="bg-gradient-to-br from-red-50 to-red-100 dark:from-red-900/20 dark:to-red-900/30 rounded-2xl p-5 text-center border border-red-200 dark:border-red-700/40 shadow-sm">
                <div class="text-3xl font-extrabold text-red-600 dark:text-red-400">{{ wrongCount }}</div>
                <div class="text-xs mt-1 font-bold uppercase tracking-wider text-red-800 dark:text-red-300">Salah</div>
              </div>
              <div class="bg-gradient-to-br from-orange-50 to-orange-100 dark:from-orange-900/20 dark:to-orange-900/30 rounded-2xl p-5 text-center border border-orange-200 dark:border-orange-700/40 shadow-sm">
                <div class="text-3xl font-extrabold text-orange-600 dark:text-orange-400">{{ accuracy }}%</div>
                <div class="text-xs mt-1 font-bold uppercase tracking-wider text-orange-800 dark:text-orange-300">Akurasi</div>
              </div>
            </div>

            <!-- CTA Lihat Pembahasan -->
            <button
              @click="goToPembahasanDetail"
              class="w-full font-extrabold py-4 px-6 rounded-2xl transition-all duration-200 flex items-center justify-center gap-3 shadow-lg active:scale-95
                     bg-tryout-primary hover:opacity-90 text-white"
            >
              <svg class="w-6 h-6" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 12h6m-6 4h6m2 5H7a2 2 0 01-2-2V5a2 2 0 012-2h5.586a1 1 0 01.707.293l5.414 5.414a1 1 0 01.293.707V19a2 2 0 01-2 2z" />
              </svg>
              <span class="text-lg">Lihat Pembahasan Detail</span>
              <svg class="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="3" d="M9 5l7 7-7 7" />
              </svg>
            </button>
          </div>

          <!-- Insight Performa -->
          <div class="rounded-3xl p-6 border-2 shadow-sm
                      bg-tryout-primary/5 border-tryout-primary/20">
            <div class="flex items-start gap-4">
              <div class="w-12 h-12 rounded-xl flex items-center justify-center flex-shrink-0 shadow-md
                          bg-tryout-primary">
                <svg class="w-7 h-7 text-white" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M13 10V3L4 14h7v7l9-11h-7z" />
                </svg>
              </div>
              <div>
                <h3 class="font-bold text-lg mb-2 text-tryout-text">💡 Insight Performa</h3>
                <p v-if="accuracy >= 80" class="text-sm font-medium leading-relaxed text-tryout-muted">
                  Luar biasa! Anda menguasai <span class="font-bold text-tryout-text">{{ accuracy }}%</span> materi. Pertahankan konsistensi ini untuk hasil maksimal di tes sebenarnya.
                </p>
                <p v-else-if="accuracy >= 60" class="text-sm font-medium leading-relaxed text-tryout-muted">
                  Bagus! Anda sudah memahami <span class="font-bold text-tryout-text">{{ accuracy }}%</span> materi. Fokus pelajari pembahasan pada topik yang masih salah.
                </p>
                <p v-else class="text-sm font-medium leading-relaxed text-tryout-muted">
                  Tetap semangat! Anda memahami <span class="font-bold text-tryout-text">{{ accuracy }}%</span> materi. Wajib pelajari pembahasan detail untuk menguasai konsep yang terlewat.
                </p>
              </div>
            </div>
          </div>
        </div>

        <!-- ── Kolom Kanan: Leaderboard ── -->
        <div class="lg:col-span-1">
          <div class="rounded-3xl shadow-2xl p-6 sticky top-8
                      bg-tryout-surface border border-tryout-border">

            <div class="flex items-center justify-between mb-6">
              <h2 class="text-xl font-extrabold text-tryout-text flex items-center gap-2">
                🏆 Leaderboard
              </h2>
              <span v-if="totalParticipants > 0"
                    class="text-xs font-bold px-3 py-1 rounded-full
                           bg-tryout-hover text-tryout-muted border border-tryout-border">
                {{ totalParticipants }} Peserta
              </span>
            </div>

            <!-- Loading leaderboard -->
            <div v-if="loadingLeaderboard" class="text-center py-10">
              <div class="animate-spin rounded-full h-8 w-8 border-b-2 border-tryout-primary mx-auto"></div>
              <p class="mt-4 font-bold text-sm text-tryout-primary">Memuat ranking nasional...</p>
            </div>

            <!-- List ranking -->
            <div v-else-if="rankings.length > 0"
                 class="space-y-3 max-h-[600px] overflow-y-auto pr-2 custom-scrollbar">
              <div
                v-for="(item, index) in rankings"
                :key="item.userId"
                :class="[
                  'flex items-center gap-3 p-3.5 rounded-2xl transition-all',
                  item.isCurrentUser
                    ? 'bg-tryout-primary/10 ring-2 ring-tryout-primary/40 shadow-md scale-[1.02]'
                    : 'bg-tryout-hover border border-tryout-border hover:bg-tryout-skeleton'
                ]"
              >
                <!-- Nomor rank -->
                <div class="w-10 h-10 rounded-full flex items-center justify-center font-extrabold text-sm flex-shrink-0 shadow-sm"
                     :class="getRankColor(index)">
                  {{ item.rank }}
                </div>

                <!-- Medali emoji -->
                <div v-if="index < 3" class="text-2xl flex-shrink-0">
                  {{ getRankBadge(index) }}
                </div>

                <div class="flex-1 min-w-0">
                  <p class="font-bold truncate text-sm"
                     :class="item.isCurrentUser ? 'text-tryout-primary' : 'text-tryout-text'">
                    {{ item.userName }}
                    <span v-if="item.isCurrentUser"
                          class="text-[10px] uppercase tracking-wider ml-1 px-1.5 py-0.5 rounded
                                 bg-tryout-primary/15 text-tryout-primary">
                      (Anda)
                    </span>
                  </p>
                  <p class="text-xs font-semibold"
                     :class="item.isCurrentUser ? 'text-tryout-primary' : 'text-tryout-muted'">
                    {{ item.totalPoint }} poin
                  </p>
                </div>

                <div v-if="item.isCurrentUser" class="flex-shrink-0">
                  <svg class="w-6 h-6 text-tryout-primary" fill="currentColor" viewBox="0 0 20 20">
                    <path fill-rule="evenodd" d="M10 9a3 3 0 100-6 3 3 0 000 6zm-7 9a7 7 0 1114 0H3z" clip-rule="evenodd" />
                  </svg>
                </div>
              </div>
            </div>

            <!-- Empty leaderboard -->
            <div v-else-if="!hasLeaderboardData && !loadingLeaderboard"
                 class="text-center py-12 rounded-2xl border-2 border-dashed
                        bg-tryout-hover border-tryout-border">
              <div class="w-16 h-16 rounded-full flex items-center justify-center mx-auto mb-4 shadow-sm
                          bg-tryout-surface border border-tryout-border">
                <svg class="w-8 h-8 text-tryout-subtle" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M17 20h5v-2a3 3 0 00-5.356-1.857M17 20H7m10 0v-2c0-.656-.126-1.283-.356-1.857M7 20H2v-2a3 3 0 015.356-1.857M7 20v-2c0-.656.126-1.283.356-1.857m0 0a5.002 5.002 0 019.288 0M15 7a3 3 0 11-6 0 3 3 0 016 0zm6 3a2 2 0 11-4 0 2 2 0 014 0zM7 10a2 2 0 11-4 0 2 2 0 014 0z" />
                </svg>
              </div>
              <p class="text-sm font-bold text-tryout-text">Belum ada data ranking</p>
              <p class="text-xs mt-1 text-tryout-muted">Tunggu peserta lain mengerjakan tryout ini</p>
            </div>

            <!-- Posisi user di luar top 100 -->
            <div v-if="!isUserInTop100 && userRank"
                 class="mt-4 p-5 rounded-2xl border-2
                        bg-tryout-primary/5 border-tryout-primary/20">
              <p class="text-center text-sm">
                <span class="font-extrabold uppercase tracking-wider text-xs text-tryout-text">Posisi Anda:</span><br>
                <span class="text-3xl font-extrabold my-1 block text-tryout-primary">#{{ userRank.rank }}</span>
                <span class="font-semibold px-3 py-1 rounded-full text-xs
                             bg-tryout-primary/10 text-tryout-primary border border-tryout-primary/20">
                  {{ userRank.totalPoint }} Poin
                </span>
              </p>
            </div>

            <!-- Peserta pertama -->
            <div v-else-if="rankings.length === 0 && totalParticipants === 0"
                 class="mt-4 p-5 rounded-2xl border-2
                        bg-tryout-primary/5 border-tryout-primary/20">
              <div class="text-center">
                <div class="w-16 h-16 rounded-full flex items-center justify-center mx-auto mb-3 shadow-md
                            bg-gradient-to-br from-yellow-400 to-yellow-600">
                  <span class="text-3xl">🥇</span>
                </div>
                <p class="font-extrabold text-lg mb-1 text-tryout-text">Peringkat #1</p>
                <p class="text-sm font-medium mb-3 px-3 py-1 rounded-full inline-block
                          bg-tryout-primary/10 text-tryout-primary">
                  Anda adalah peserta pertama!
                </p>
                <p class="text-xs font-semibold text-tryout-muted">Tunggu peserta lain untuk melihat ranking lengkap.</p>
              </div>
            </div>

          </div>
        </div>

      </div>
    </div>
  </main>
</template>
